### 2-4 可变参数

```python
def func(*args,**kwargs):
	print(args)  # args： 接收位置参数，返回元祖
	print(kwargs)# kwargs : 接收关键词参数，返回字典
	
func(10,20,30) #位置传参
func(a=10,b=20,c=30,d=40) #关键词传参

# 参数是一个序列  tuple
def total(*nums):
    print(nums)
    print(type(nums))
    result = 0
    for i in nums:
        result += i
    return result

result = total(1,2,3,4,5)
print(result)

# 参数是一个字典
def f(d):
    print(type(d))
    for k,v in d.items():
        print(k,'->',v)

d = {'name':'tom','age':34}

f(d)
```

## 4- 匿名函数 lambda

```python
#普通函数
def func(num1,num2):
	return num1 ** num2
	
print(func(2,3))


def add(a,b):
    return a+b

a = add(2,3)
print(a)



from functools import reduce


# lambda表达式
mul = lambda a,b: a*b

result = mul(2,3)
print(result)


list1 = [1,2,3,4,5]

# lambda表达式
# map 映射
mapResult = map(lambda x:x**2 , list1)

print(list(mapResult))

# lambda表达式
# filter 过滤
filterResult = filter(lambda x:x%2 == 0 ,list1)
print(list(filterResult))

# lambda表达式
# reduce 集合累加
reduceResult = reduce (lambda x,y: x+y , list1)
print(reduceResult)


## lambda表达式
# reduce 将list1 转成一个大整数
list2 = [1,3,40,6,8,2,0,2,4]
result = reduce(lambda x ,y : x * 10 ** len(str(y)) + y, list2)
print(result)

```



## 5- 函数的作用域

- 不可变数据类型在函数中使用时需要线 global;
- 可变数据类型在函数中使用时不需要线 global;

```python

# 定义全局变量
num1 = 10   # 不可变数据类型
list1 = [1,2,3,4,5]  # 可变数据类型

def f():
    global num1     # 不可变数据类型在函数中使用时需要线 global;
    num1 = 20
    list1[2] = 8    # 可变数据类型在函数中使用时不需要线 global;
    print('在函数中打印 num1 ==',num1)
    print('在函数中打印 list1 ==',list1)


f()
print('在函数外打印 num1 ==',num1)
print('在函数外打印 list1 ==',list1)



```
